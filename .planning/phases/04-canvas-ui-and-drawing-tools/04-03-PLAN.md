**onChange handler (local -> Y.Array):**

```typescript
const handleChange = useCallback(
  (elements: readonly ExcalidrawElement[]) => {
    if (isApplyingRemoteRef.current) return // skip when applying remote update

    ydoc.transact(() => {
      yElements.delete(0, yElements.length)
      yElements.insert(0, elements as unknown[])
    })
  },
  [ydoc, yElements],
)
```

The transaction origin defaults to `null` (not `'remote'` or `'init'`), so the ydoc update handler in CollabProvider will relay it to the socket.

**yElements observer (remote -> Excalidraw):**

Set up in a `useEffect`:

```typescript
useEffect(() => {
  const excalidrawAPI = excalidrawAPIRef.current

  const observer = () => {
    if (!excalidrawAPIRef.current) return
    isApplyingRemoteRef.current = true
    excalidrawAPIRef.current.updateScene({
      elements: yElements.toArray() as ExcalidrawElement[],
      captureUpdate: CaptureUpdateAction.NEVER,
    })
    // Use requestAnimationFrame or setTimeout(0) to reset the flag after React processes the update
    requestAnimationFrame(() => {
      isApplyingRemoteRef.current = false
    })
  }

  yElements.observe(observer)
  return () => yElements.unobserve(observer)
}, [yElements])
```

CRITICAL: `CaptureUpdateAction.NEVER` prevents remote changes from polluting the local undo/redo stack (CANV-07 requirement).

**IMPORTANT: The observer fires for ALL Y.Array changes, including local ones.** To prevent the infinite loop (onChange -> Y.Array change -> observer -> updateScene -> onChange), the `isApplyingRemoteRef` flag is set to `true` before `updateScene` and reset after. The `handleChange` callback checks this flag and skips if true.

An alternative approach: Instead of the ref flag, only call `updateScene` from the observer when the change origin is `'remote'`. To do this, listen on `ydoc.on('update', (update, origin) => { if (origin === 'remote') { /* read yElements and updateScene */ } })` instead of using `yElements.observe`. Choose whichever approach is cleaner — both prevent the infinite loop.

**excalidrawAPI callback:**

```typescript
const excalidrawAPIRef = useRef<ExcalidrawImperativeAPI | null>(null)

const handleAPI = useCallback(
  (api: ExcalidrawImperativeAPI) => {
    excalidrawAPIRef.current = api
    setExcalidrawAPI(api)
  },
  [setExcalidrawAPI],
)
```

**Architecture:** The page has an outer component (handles join-first flow) and an inner component (wrapped in CollabProvider, consumes useCollab):

```typescript
'use client'

import { useEffect, useState } from 'react'
import { useParams } from 'next/navigation'
import dynamic from 'next/dynamic'
import Link from 'next/link'
import { Button } from '@/components/ui/button'
import { DoorClosed, Loader2 } from 'lucide-react'
import { CanvasHeader } from '@/src/components/canvas/canvas-header'
import { CollabProvider, useCollab } from '@/src/contexts/collab-context'

const ExcalidrawWrapper = dynamic(
  () => import('@/src/components/canvas/excalidraw-wrapper'),
  { ssr: false }
)

type JoinState = 'joining' | 'joined' | 'not-found'

export default function CanvasPage() {
  const { slug } = useParams<{ slug: string }>()
  const [joinState, setJoinState] = useState<JoinState>('joining')

  useEffect(() => {
    const joinRoom = async () => {
      try {
        const res = await fetch(`/api/rooms/${slug}/join`, {
          method: 'POST',
          credentials: 'include',
          headers: { 'Content-Type': 'application/json' },
        })
        if (!res.ok) {
          setJoinState('not-found')
          return
        }
        setJoinState('joined')
      } catch {
        setJoinState('not-found')
      }
    }
    joinRoom()
  }, [slug])

  if (joinState === 'not-found') {
    return (
      <div className="flex min-h-screen flex-col items-center justify-center gap-4 bg-background">
        <DoorClosed className="h-12 w-12 text-muted-foreground" />
        <h1 className="text-xl font-semibold text-foreground">Room not found</h1>
        <p className="max-w-xs text-center text-sm text-muted-foreground">
          This room doesn&apos;t exist or the link may have expired.
        </p>
        <Button asChild variant="default">
          <Link href="/feed">Back to UniShare</Link>
        </Button>
      </div>
    )
  }

  if (joinState === 'joining') {
    return (
      <div className="fixed inset-0 z-20 flex flex-col items-center justify-center gap-6 bg-background"
        role="status" aria-live="polite" aria-label="Connecting to collaboration room">
        <Loader2 className="h-6 w-6 animate-spin text-primary" />
        <p className="text-sm font-semibold text-foreground">Connecting to room...</p>
      </div>
    )
  }

  // Join succeeded — mount CollabProvider which opens socket
  return (
    <CollabProvider slug={slug}>
      <CanvasInner />
    </CollabProvider>
  )
}

function CanvasInner() {
  const { connectionStatus } = useCollab()

  if (connectionStatus === 'connecting') {
    return (
      <div className="fixed inset-0 z-20 flex flex-col items-center justify-center gap-6 bg-background"
        role="status" aria-live="polite" aria-label="Connecting to collaboration room">
        <Loader2 className="h-6 w-6 animate-spin text-primary" />
        <p className="text-sm font-semibold text-foreground">Connecting to room...</p>
      </div>
    )
  }

  return (
    <div className="flex h-screen flex-col">
      <CanvasHeader />
      <main className="relative flex-1">
        <ExcalidrawWrapper />
      </main>
    </div>
  )
}
```

This ensures:

- Excalidraw does NOT mount until `connectionStatus === 'connected'` (loading overlay covers it)
- CollabProvider opens the socket only after join succeeds (no socket attempt on 404)
- `initialElements` from `room-joined` are available when Excalidraw mounts (passed via context)
